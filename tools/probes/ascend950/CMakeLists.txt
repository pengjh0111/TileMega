function(as1_asc_target name source)
  set_source_files_properties("${source}" PROPERTIES LANGUAGE ASC)
  add_executable(${name} EXCLUDE_FROM_ALL "${source}")
  target_include_directories(${name} PRIVATE "${CATLASS_ROOT}/include" "${CATLASS_ROOT}/examples/common" "${CANN}/include/aclnn")
  target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../include")
  target_compile_definitions(${name} PRIVATE CATLASS_ARCH=3510 _GLIBCXX_USE_CXX11_ABI=1)
  target_compile_options(${name} PRIVATE --npu-arch=dav-3510 -Wno-macro-redefined -Wno-ignored-attributes -Wno-unused-value)
  target_link_options(${name} PRIVATE -Wno-unused-command-line-argument)
  target_link_directories(${name} PRIVATE "${CANN}/lib64" "${CANN}/x86_64-linux/lib64" "${CANN}/x86_64-linux/devlib")
  target_link_libraries(${name} PRIVATE ascendc_runtime runtime profapi mmpa ascend_dump c_sec error_manager ascendcl tiling_api nnopbase platform dl ascendalog unified_dlog ascend_hal m)
endfunction()

# Preserve the pinned checkout. Only host teardown and the resource request change.
function(as1_safe_sample name relative_source)
  file(READ "${CATLASS_ROOT}/${relative_source}" original)
  string(FIND "${original}" "ACL_CHECK(aclrtResetDevice(options.deviceId));" reset_pos)
  if(reset_pos LESS 0)
    message(FATAL_ERROR "Sample teardown changed; review the pinned source")
  endif()
  string(REPLACE "ACL_CHECK(aclrtResetDevice(options.deviceId));" "/* AS-1: process-scoped aclFinalize below; no reset call. */" safe "${original}")
  set(core_query "auto aicCoreNum = platform_ascendc::PlatformAscendCManager::GetInstance()->GetCoreNumAic();")
  string(FIND "${safe}" "${core_query}" query_pos)
  if(query_pos LESS 0)
    message(FATAL_ERROR "Sample core query changed; review its resource request")
  endif()
  string(REPLACE "${core_query}" "${core_query}\n    aicCoreNum = aicCoreNum < 2 ? aicCoreNum : 2;" safe "${safe}")
  set(generated "${CMAKE_CURRENT_BINARY_DIR}/${name}.cpp")
  file(WRITE "${generated}" "${safe}")
  as1_asc_target(${name} "${generated}")
endfunction()
