# Preserve the pinned checkout. Only host teardown and the resource request change.
function(as1_safe_sample name relative_source)
  file(READ "${CATLASS_ROOT}/${relative_source}" original)
  string(FIND "${original}" "ACL_CHECK(aclrtResetDevice(options.deviceId));" reset_pos)
  if(reset_pos LESS 0)
    message(FATAL_ERROR "Sample teardown changed; review the pinned source")
  endif()
  string(REPLACE "ACL_CHECK(aclrtResetDevice(options.deviceId));" "/* AS-1: process-scoped aclFinalize below; no reset call. */" safe "${original}")
  set(core_query "auto aicCoreNum = platform_ascendc::PlatformAscendCManager::GetInstance()->GetCoreNumAic();")
  string(FIND "${safe}" "${core_query}" query_pos)
  if(query_pos LESS 0)
    message(FATAL_ERROR "Sample core query changed; review its resource request")
  endif()
  string(REPLACE "${core_query}" "${core_query}\n    aicCoreNum = aicCoreNum < 2 ? aicCoreNum : 2;" safe "${safe}")
  set(generated "${CMAKE_CURRENT_BINARY_DIR}/${name}.cpp")
  file(WRITE "${generated}" "${safe}")
  as1_asc_target(${name} "${generated}")
endfunction()
